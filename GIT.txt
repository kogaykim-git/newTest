git branch -M main - создание main ветки
mkdir <имя папки> - создание новой папки
git init - инициализация гита
cd - change directory смена директории
ls - list список папок и файлов
rm - remove удаление файлов и папок зависит от ключей(аргументов)
rm - rf <directory name> - remove recursive force удаление файлов и папок рекурсивное
ls -l - list "длинный список" список папок и файлов с подробной информацией
git add origin <ссылка на репозиторий> - присваивание к origin ссылки удаленного репозитория
git add <file_name> - отправка на staging
git commit -m "message name" - создание коммита относительно staging
git push origin main - отправка коммитов из ветки main на удаленный репозиторий origin
git status - просмотр текущего состояния рабочего каталога и staging area (отслеживаемые/неотслеживаемые файлы, изменённые файлы и готовые к коммит

git branch <наименование ветки> - создание новой ветки
git switch <название ветки> - переключение на определенную ветку
git merge <название ветки> - слияние ветки в ту в которой мы находимся
git branch -d <название ветки> - удалить ветку
git branch -D <название ветки> - принудительно удалить ветку
